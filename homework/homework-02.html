FINAL_TIME = 5.0

def qoi_solve_ivp(k, c):
    def rhs(t, state):
        y, velocity = state
        return [velocity, -c * velocity - k * y]
    solution = solve_ivp(
        rhs, (0.0, FINAL_TIME), [1.0, 0.0],
        t_eval=[FINAL_TIME], rtol=1e-10, atol=1e-12,
    )
    return solution.y[0, -1]

def qoi_closed_form(k, c):
